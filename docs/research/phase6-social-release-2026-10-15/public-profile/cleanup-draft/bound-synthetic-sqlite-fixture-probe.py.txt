def snapshot(db: sqlite3.Connection) -> dict[str, object]:

    q='('+','.join('?' for _ in targets)+')'
    return {
      'epoch':db.execute('SELECT epoch FROM main_store_authority WHERE singleton=1').fetchone()[0],
      'sessions':db.execute(f'SELECT count(*) FROM sessions WHERE public_id IN {q}',targets).fetchone()[0],
      'actions':db.execute(f'SELECT count(*) FROM action_receipts WHERE sender IN {q}',targets).fetchone()[0],
      'wallets':db.execute(f'SELECT count(*) FROM wallet_effects WHERE public_id IN {q}',targets).fetchone()[0],
      'helpers':db.execute("SELECT count(*) FROM sqlite_master WHERE type='table' AND name GLOB ?",(PREFIX+'*',)).fetchone()[0],
      'collections':dict(db.execute("SELECT name,value FROM collections WHERE name IN ('social','civic','growth','root:growth')")),
      'entries':db.execute('SELECT coll,map,key,ord,ix,tx,jx,value FROM entries ORDER BY coll,map,key').fetchall(),
      'world':db.execute('SELECT seq,name,text FROM world_shards ORDER BY seq').fetchall(),
      'world_meta':db.execute('SELECT key,value FROM world_meta ORDER BY key').fetchall(),
      'unrelated_session':db.execute("SELECT value FROM sessions WHERE public_id='55555555-5555-4555-8555-555555555555'").fetchone()[0],

    }

def execute_atomic(db: sqlite3.Connection, sql: str, expect_error: bool) -> str | None:
    try:
        db.executescript('BEGIN IMMEDIATE;\n'+sql+'\nCOMMIT;')
    except sqlite3.Error as error:
        db.rollback()
        if not expect_error:
            raise
        return str(error)
    if expect_error:
        raise AssertionError('rehearsal unexpectedly committed')
    return None

def scalar(db: sqlite3.Connection, sql: str, params: tuple[object,...]=()) -> object:
    return db.execute(sql,params).fetchone()[0]
