-- READ-ONLY SUPPLEMENTAL INVENTORY DRAFT. DO NOT RUN AS PART OF CLEANUP.
-- Count-only coverage for reference surfaces the original value-focused inventory did not prove.
WITH targets(id) AS (VALUES
  ('bfcb8ef0-4d00-4d8e-b665-bb8b11d45e56'),
  ('453a0f63-5b90-4e7b-b730-4ea9ff54b70d'),
  ('6cea4961-b7fc-4bb7-a6db-a28e9327014f')
), entry_key_refs AS (
  SELECT DISTINCT e.coll,e.map,e.key
  FROM entries e,targets t
  WHERE instr(e.key,t.id)>0
), entry_key_only_refs AS (
  SELECT DISTINCT e.coll,e.map,e.key
  FROM entries e,targets t
  WHERE instr(e.key,t.id)>0
    AND e.key<>t.id
    AND instr(e.value,t.id)=0
    AND instr(COALESCE(e.jx,''),t.id)=0
), collection_name_refs AS (
  SELECT DISTINCT c.name
  FROM collections c,targets t
  WHERE instr(c.name,t.id)>0
)
SELECT json_object(
  'entry_key_refs',(SELECT count(*) FROM entry_key_refs),
  'previously_uncaught_entry_key_only_refs',(SELECT count(*) FROM entry_key_only_refs),
  'collection_name_refs',(SELECT count(*) FROM collection_name_refs)
) AS supplemental_reference_summary;
