import assert from 'node:assert/strict'
import { createHash } from 'node:crypto'
import { mkdtempSync, readFileSync, readdirSync, realpathSync, rmSync, statSync } from 'node:fs'
import { join, relative } from 'node:path'
import { tmpdir } from 'node:os'
import { spawnSync } from 'node:child_process'
import { fileURLToPath } from 'node:url'

// Copy this file to actual-package.mjs beside it, then run it from the repository root after the accepted build.
const root = fileURLToPath(new URL('../../../..', import.meta.url))
const sourceSha = '5bb46a442a2b6d1236b988a67d09c653f61ea766'
const folder = realpathSync(mkdtempSync(join(tmpdir(), 'allworld-actual-sealed-')))
const packager = join(root, 'scripts/package-joinallworld.mjs'), tooling = join(root, 'deploy/tooling')
const sha = bytes => createHash('sha256').update(bytes).digest('hex')
function run(name, runId) {
  const output = join(folder, name)
  const result = spawnSync(process.execPath, [packager, root, output, sourceSha, 'false', tooling], { encoding: 'utf8', env: { PATH: process.env.PATH, GITHUB_ACTIONS: 'true', GITHUB_RUN_ID: runId, GITHUB_RUN_ATTEMPT: '2' } })
  assert.equal(result.status, 0, result.stderr)
  return { output, receipt: JSON.parse(result.stdout) }
}
function files(rootPath) {
  const found = new Map(); const walk = path => { for (const name of readdirSync(path).sort()) { const full = join(path, name), stat = statSync(full); if (stat.isDirectory()) walk(full); else found.set(relative(rootPath, full), sha(readFileSync(full))); } }; walk(rootPath); return found
}
try {
  const first = run('same-a', '400000001'), second = run('same-b', '400000001'), changed = run('changed', '400000002')
  assert.equal(first.receipt.packageDigest, second.receipt.packageDigest)
  const a = files(first.output), b = files(second.output), c = files(changed.output)
  assert.deepEqual(a, b)
  const changedNames = [...a].filter(([name, digest]) => c.get(name) !== digest).map(([name]) => name)
  assert.deepEqual(changedNames, ['manifest.json', 'worker.js'])
  const workerBytes = statSync(join(first.output, 'worker.js')).size
  assert.ok(workerBytes <= 3_500_000)
  console.log(JSON.stringify({ passed: true, fileCount: a.size, packageDigest: first.receipt.packageDigest, workerBytes, sameRunIdentical: true, changedRunFiles: changedNames }, null, 2))
} finally { rmSync(folder, { recursive: true, force: true }) }
